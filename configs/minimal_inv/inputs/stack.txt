% Stack macro syntax: define <name> = <number | @<arithmetic-expression>@>
% All Z values below are derived from these definitions.  Changing a macro
% moves every dependent layer; gen_sde.py rejects controlled interfaces that
% lose their positive overlap.

% Shared vertical interface tolerance (um).  A zero value denotes a valid
% shared-face landing; a negative gap is rejected by gen_sde.py.
define contact_overlap = 0.000

% Device and local-front-metal stack.
define gate_z0 = 0.000
define lower_sd_contact_z = 0.006
define md_sd_z0 = 0.112
% Must stay >= upper-sheet-top + high_k_thickness; validated by gen_sde.py.
define md_gate_z0 = 0.126
% Gate terminates exactly where its MD contact begins: no residual Gate lump.
define gate_z1 = $md_gate_z0
define md_z1 = 0.140
define mrw_z0 = $lower_sd_contact_z
define mrw_z1 = $md_z1
define v0_z0 = @<md_z1-contact_overlap>@
define v0_z1 = @<v0_z0+0.030>@
define m0_z0 = @<v0_z1-contact_overlap>@
define m0_z1 = @<m0_z0+0.030>@

% Higher frontside routing stack: each via overlaps its two metal landings.
define v1_z0 = @<m0_z1-contact_overlap>@
define v1_z1 = @<v1_z0+0.030>@
define m1_z0 = @<v1_z1-contact_overlap>@
define m1_z1 = @<m1_z0+0.030>@
define v2_z0 = @<m1_z1-contact_overlap>@
define v2_z1 = @<v2_z0+0.030>@
define m2_z0 = @<v2_z1-contact_overlap>@
define m2_z1 = @<m2_z0+0.030>@
define v3_z0 = @<m2_z1-contact_overlap>@
define v3_z1 = @<v3_z0+0.030>@
define m3_z0 = @<v3_z1-contact_overlap>@
define m3_z1 = @<m3_z0+0.030>@

% Backside routing stack. BMD touches lower SD at lower_sd_contact_z;
% BV0 overlaps BMD and BM0 by contact_overlap.
define bmd_z0 = -0.004
define bmd_z1 = $lower_sd_contact_z
% BMD gate contacts stop at the gate underside, making BMD the downward
% extension of the gate rather than replacing part of its volume.
define bmd_gate_z1 = $gate_z0
define bm0_z1 = -0.034
define bm0_z0 = @<bm0_z1-0.030>@
define bv0_z0 = @<bm0_z1-contact_overlap>@
define bv0_z1 = @<bmd_z0+contact_overlap>@
