% ---------------------------------------------------------------------------
% DRC syntax: rule <Layer.Type.Number> <ERROR|WARNING|INFO> <check> key=value
% Units are um.  The rule declares a limit; gen_drc_report.py calculates the
% measured layout value and reports measured / requirement / margin.
%
% Supported checks: min_width, min_area, min_spacing, enclosure, z_landing,
% contact_recipe.  contact_recipe shares gen_sde.py's gate/spacer/planned-SD
% construction, so its MD/BMD decision matches the generated 3-D structure.

rule GATE.W.001       ERROR   min_width      layer=7  axis=x  min=0.012
rule GATE.L.001       ERROR   min_width      layer=7  axis=y  min=0.040
rule GATE.S.001       ERROR   min_spacing    layer=7  axis=x  min=0.030

rule MD.W.001         ERROR   min_width      layer=16 axis=x  min=0.012
rule BMD.W.001        ERROR   min_width      layer=51 axis=x  min=0.012
rule V0.W.001         ERROR   min_width      layer=18 axis=x  min=0.010
rule V0.L.001         ERROR   min_width      layer=18 axis=y  min=0.010

% Every front/back via must be enclosed by its landing metal in the XY plane.
rule MD.V0.EN.001     ERROR   enclosure      outer=16 inner=18 min=0.001
rule BMD.BV0.EN.001   ERROR   enclosure      outer=51 inner=52 min=0.001

% Through-stack landings: zero is accepted when the layers share a face.
rule V0.Z.001         ERROR   z_landing      layer=18 lower=16 upper=19 min=0.000
rule BV0.Z.001        ERROR   z_landing      layer=52 lower=53 upper=51 min=0.000

% MD/BMD are classified in the order SD > Gate > Other from their XY masks.
% The S/D check verifies a real Z contact.  The gate edge checks guarantee
% MD ends at/above gate top and BMD ends at/below gate bottom respectively.
rule MD.RECIPE.001    ERROR   contact_recipe layer=16 sd=upper_sd gate=7 side=top    gate_z0=$md_gate_z0
rule BMD.RECIPE.001   ERROR   contact_recipe layer=51 sd=lower_sd gate=7 side=bottom gate_z1=$bmd_gate_z1
